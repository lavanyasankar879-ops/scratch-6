    <script>
        const canvas = document.getElementById('scratchCanvas');
        const ctx = canvas.getContext('2d');
        const container = document.getElementById('cardBox');
        let isDrawing = false;

        function initScratchCard() {
            canvas.width = container.offsetWidth;
            canvas.height = container.offsetHeight;

            // Generate an elegant champagne-gold metallic linear gradient layout
            const goldGrad = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            goldGrad.addColorStop(0, '#e5c158');   // Bright champagne highlight
            goldGrad.addColorStop(0.2, '#f3db8b'); // Soft reflection
            goldGrad.addColorStop(0.5, '#d4af37'); // Classic gold base
            goldGrad.addColorStop(0.8, '#f3db8b'); // Shimmer accent
            goldGrad.addColorStop(1, '#b5942b');   // Deep gold shadows
            
            ctx.fillStyle = goldGrad;
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Subtle textured golden glitter dust particle simulation overlay
            for (let i = 0; i < 350; i++) {
                ctx.fillStyle = 'rgba(255, 255, 255, 0.45)';
                ctx.fillRect(Math.random() * canvas.width, Math.random() * canvas.height, 1.5, 1.5);
            }

            // Decorative gold border frames printed on top of the foil coating cover sheet
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.35)';
            ctx.lineWidth = 2;
            ctx.strokeRect(15, 15, canvas.width - 30, canvas.height - 30);

            // Instructions label layout typography execution logic block
            ctx.font = "italic bold 18px Georgia";
            ctx.fillStyle = "#684c0c"; // Deep bronze contrast shadow
            ctx.textAlign = "center";
            ctx.textBaseline = "middle";
            ctx.fillText("✨ Scratch with Love ✨", canvas.width / 2, canvas.height / 2);
            
            // Set operational blend status context mode to clear matching pixels when brushed
            ctx.globalCompositeOperation = 'destination-out';
        }

        // Consolidated drawing execution script block handling calculations flawlessly
        function scratch(e) {
            if (!isDrawing) return;
            
            const rect = canvas.getBoundingClientRect();
            // Supports touch interface arrays concurrently alongside standard point arrays 
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;
            
            const x = clientX - rect.left;
            const y = clientY - rect.top;

            ctx.lineWidth = 45; // Broad responsive erasing brush parameters
            ctx.lineCap = 'round';
            ctx.lineTo(x, y);
            ctx.stroke();
            ctx.beginPath();
            ctx.moveTo(x, y);
        }

        // Web Desktop Event Interfaces
        canvas.addEventListener('mousedown', (e) => { isDrawing = true; ctx.beginPath(); scratch(e); });
        canvas.addEventListener('mousemove', scratch);
        window.addEventListener('mouseup', () => { isDrawing = false; });

        // Mobile Handset High-Precision Touch Interface Mapping
        canvas.addEventListener('touchstart', (e) => { 
            isDrawing = true; 
            ctx.beginPath(); 
            scratch(e); 
        });
        canvas.addEventListener('touchmove', (e) => {
            // Stops ambient mobile scrolling gestures during canvas scratching manipulation
            e.preventDefault(); 
            scratch(e);
        });
        window.addEventListener('touchend', () => { isDrawing = false; });

        // Compile layout profiles seamlessly upon window execution trigger cycles
        window.onload = initScratchCard;
    </script>
